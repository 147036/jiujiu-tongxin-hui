  opacity: 1; transform: none; /* 默认可见 */
}
.section-block.animate-in {
  opacity: 0; transform: translateY(38px); /* 动画初始状态 */
}
.section-block.animate-in.in-view {
  opacity: 1; transform: none; /* 触发后显示 */
}
